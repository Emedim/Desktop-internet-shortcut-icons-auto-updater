В качестве одного оператора int может быть объявлена и пееременная типа int и указаетль на тип int:
int a = 10, *b = &a;        // a – переменная типа int, b – указатель на int.

const int *ptr;             //ptr – изменять значение по адресу через этот указатель нельзя
int * const ptr;            //ptr – нельзя изменть адрес, на который указывает указатель
const int * const ptr;      //нельзя ни менять адрес, ни менять значение по адресу

Строки
char s[] = "string";        //обычный массив, можно менять значение строки
const char *s = "string";   //указывает на неизменяемую область памяти
char *s = "string";         //всё-равно нельзя изменить строчку
